1//! COMTRADE 文件组写出。
2
3use std::path::Path;
4
5use cbase::error::{Error, FileRole, Result};
6use cff::CffFile;
7use cfg::DataType;
8
9use super::Comtrade;
10
11impl Comtrade {
12    /// 整组写出到目录
13    pub fn write_to_dir(&self, dir: &Path, stem: &str, dt: DataType) -> Result<()> {
14        std::fs::create_dir_all(dir).map_err(|e| Error::Io {
15            path: dir.to_path_buf(),
16            source: e,
17        })?;
18
19        // 同步目标数据格式，保证 CFG 声明与 DAT 字节一致
20        let mut cfg = self.config.clone();
21        cfg.data_type = dt;
22
23        // CFG
24        let cfg_path = dir.join(format!("{}.cfg", stem));
25        cfg.write_file(&cfg_path)?;
26
27        // DAT
28        if let Some(ref dat) = self.data {
29            let dat_path = dir.join(format!("{}.dat", stem));
30            dat.write_file(&dat_path, &cfg, dt)?;
31        }
32
33        // INF（可选）
34        if let Some(ref inf) = self.inf {
35            let inf_path = dir.join(format!("{}.inf", stem));
36            inf.write_file(&inf_path)?;
37        }
38
39        // DMF（可选）
40        if let Some(ref dmf) = self.dmf {
41            let dmf_path = dir.join(format!("{}.dmf", stem));
42            dmf.write_file(&dmf_path)?;
43        }
44
45        Ok(())
46    }
47
48    /// 编码为 CFF 单文件字节流（CFG+INF+DAT，丢弃 HDR/DMF，与 `CffFile::encode` 一致）。
49    pub fn to_cff_bytes(&self, dt: DataType) -> Result<Vec<u8>> {
50        let data = self
51            .data
52            .as_ref()
53            .ok_or_else(|| Error::parse(FileRole::Dat, 0, "文件不含采样数据"))?;
54        Ok(CffFile::encode(&self.config, data, self.inf.as_ref(), dt))
55    }
56
57    /// 写出为 CFF 单文件。
58    pub fn write_cff(&self, path: &Path, dt: DataType) -> Result<()> {
59        let bytes = self.to_cff_bytes(dt)?;
60        std::fs::write(path, bytes).map_err(|e| Error::Io {
61            path: path.to_path_buf(),
62            source: e,
63        })
64    }
65}
