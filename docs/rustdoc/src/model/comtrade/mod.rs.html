35/// 通道数据视图（零拷贝）
36#[derive(Debug, Clone)]
37pub struct ChannelView<'a> {
38    /// 通道定义（只读引用）
39    pub definition: &'a cfg::AnalogChannel,
40    /// 该通道的采样数据（只读引用）
41    pub samples: &'a [f64],
42}
43
44/// 编辑操作中标识通道种类
45#[derive(Debug, Clone, Copy, PartialEq, Eq)]
46pub enum ChannelKind {
47    /// 模拟量通道
48    Analog,
49    /// 状态量通道
50    Status,
51}

53/// COMTRADE 数据模型
54#[derive(Debug, Clone, Default)]
55pub struct Comtrade {
56    /// CFG 配置（必需）
57    pub config: Config,
58    /// DAT 采样数据（必需）
59    pub data: Option<DatFile>,
60    /// HDR 头文件（仅记录路径，未解析）
61    pub hdr: Option<HdrFile>,
62    /// INF 信息文件（可选）
63    pub inf: Option<InfFile>,
64    /// DMF 设备模型文件（可选）
65    pub dmf: Option<DmfFile>,
66    /// 设备拓扑（由 DMF 优先，否则 INF 构建）
67    pub equipment: EquipmentGroup,
68    /// 文件组路径信息
69    pub paths: ComtradePaths,
70}
71
72impl Comtrade {
73    /// 数据访问：获取模拟通道
74    pub fn analog_channel(&self, index: usize) -> Option<ChannelView<'_>> {
75        let def = self.config.analog(index)?;
76        let data = self.data.as_ref()?;
77        let samples = data.analog_samples(index - 1)?;
78        Some(ChannelView {
79            definition: def,
80            samples,
81        })
82    }
83
84    /// 变位检测
85    pub fn changed_statuses(&self) -> Vec<(usize, Vec<StatusChange>)> {
86        if let Some(ref dat) = self.data {
87            dat.changed_statuses(Some(self.config.start_time))
88        } else {
89            Vec::new()
90        }
91    }
92}
