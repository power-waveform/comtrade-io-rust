8/// 形状对齐：按 CFG 声明的通道数与采样点数截断/补齐（对齐 Python _validate_shape）
9pub(crate) fn fit_to_config(mut dat: DatFile, cfg: &Config) -> Result<DatFile> {
10    let actual_rows = dat.len();
11    let expected_rows = cfg
12        .sampling
13        .segments
14        .last()
15        .map(|s| s.end_point)
16        .unwrap_or(0);
17
18    if actual_rows > expected_rows && expected_rows > 0 {
19        // 截断
20        dat.sample_index.truncate(expected_rows);
21        dat.timestamp_us.truncate(expected_rows);
22        for col in &mut dat.analogs {
23            col.truncate(expected_rows);
24        }
25        for col in &mut dat.statuses {
26            col.truncate(expected_rows);
27        }
28    }
29
30    // 校验列数
31    let actual_analog = dat.analog_count();
32    if actual_analog < cfg.channels.analog {
33        // 模拟通道不足
34        if actual_analog == 0 && cfg.channels.analog > 0 {
35            return Err(Error::parse(
36                FileRole::Dat,
37                0,
38                format!(
39                    "期望最少{}列模拟量，实际{}列",
40                    cfg.channels.analog + 2,
41                    actual_analog + 2
42                ),
43            ));
44        }
45        // 补零列
46        for _ in actual_analog..cfg.channels.analog {
47            dat.analogs.push(vec![0.0; dat.len()]);
48        }
49    }
50
51    let actual_status = dat.status_count();
52    if actual_status < cfg.channels.status {
53        // 补零列
54        for _ in actual_status..cfg.channels.status {
55            dat.statuses.push(vec![0u8; dat.len()]);
56        }
57    }
58
59    // 截断多余列
60    dat.analogs.truncate(cfg.channels.analog);
61    dat.statuses.truncate(cfg.channels.status);
62
63    Ok(dat)
64}

66/// 应用 timemult 时间倍乘系数
67pub(crate) fn apply_timemult(dat: &mut DatFile, cfg: &Config) {
68    let timemult = cfg.timemult;
69    if (timemult - 1.0).abs() > f64::EPSILON {
70        for ts in &mut dat.timestamp_us {
71            *ts *= timemult;
72        }
73    }
74}
