19/// 文件头
20#[derive(Debug, Clone)]
21pub struct Header {
22    /// 站名（录波装置所在变电站/站点名称）
23    pub station: String,
24    /// 录波装置名称
25    pub recorder: String,
26    /// 录波数据标准版本（1991/1999/2001/2008/2013/2017）
27    pub version: Version,
28}
29
30impl Header {
31    fn from_line(line: &str) -> Header {
32        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
33        let station = parts.first().unwrap_or(&"").to_string();
34        let recorder = parts.get(1).unwrap_or(&"").to_string();
35        // 六个版本号全部识别；头部无版本字段或版本号无法识别时回落 1991
36        // （对齐 Python `Version.from_value(str_arr[2], Version.V1991)`）。
37        // 原实现只匹配 "1999"，导致 2001/2008/2013/2017 的文件被静默判成 1991。
38        let version = parts
39            .get(2)
40            .and_then(|s| Version::parse(s))
41            .unwrap_or_default();
42        Header {
43            station,
44            recorder,
45            version,
46        }
47    }
48
49    fn to_line(&self) -> String {
50        format!(
51            "{},{},{}",
52            self.station,
53            self.recorder,
54            self.version.as_str()
55        )
56    }
57}

59/// 通道计数
60#[derive(Debug, Clone, Copy)]
61pub struct ChannelCount {
62    /// 通道总数（模拟量 + 状态量）
63    pub total: usize,
64    /// 模拟量通道数
65    pub analog: usize,
66    /// 状态量通道数
67    pub status: usize,
68}
69
70impl ChannelCount {
71    fn from_line(line: &str) -> Result<ChannelCount> {
72        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
73        if parts.len() < 3 {
74            return Err(Error::parse(FileRole::Cfg, 2, "通道数行格式错误"));
75        }
76        let total = parts[0]
77            .parse::<usize>()
78            .map_err(|_| Error::number(FileRole::Cfg, parts[0].to_string()))?;
79        let analog = extract_digits(parts.get(1).unwrap_or(&"0"))
80            .map_err(|_| Error::number(FileRole::Cfg, parts[1].to_string()))?;
81        let status = extract_digits(parts.get(2).unwrap_or(&"0"))
82            .map_err(|_| Error::number(FileRole::Cfg, parts[2].to_string()))?;
83        Ok(ChannelCount {
84            total,
85            analog,
86            status,
87        })
88    }
89
90    fn to_line(self) -> String {
91        format!("{},{}A,{}D", self.total, self.analog, self.status)
92    }
93}

180impl Config {
181    /// 从字符串解析 CFG 配置
182    #[allow(clippy::should_implement_trait)]
183    pub fn from_str(text: &str) -> Result<Config> {
184        let lines: Vec<&str> = text
185            .lines()
186            .map(|line| line.trim_end_matches('\r'))
187            .filter(|line| !line.is_empty())
188            .collect();
189
190        if lines.len() < 10 {
191            return Err(Error::parse(
192                FileRole::Cfg,
193                0,
194                format!("行数不足，至少需要10行，实际{}行", lines.len()),
195            ));
196        }
197
198        let cursor = 0;
199
200        // 第1行: 文件头
201        let header = Header::from_line(lines[cursor]);
202
203        // 第2行: 通道数量
204        let channels = ChannelCount::from_line(lines[cursor + 1])?;
205
206        // 跳过通道信息行，处理采样信息
207        let sampling_start = channels.total + 2;
208        let mut cursor = sampling_start;
209        if cursor >= lines.len() {
210            return Err(Error::parse(FileRole::Cfg, 0, "行数不足"));
211        }
212
213        // 采样频率
214        let freq: f64 = lines[cursor].parse().unwrap_or(50.0);
215        cursor += 1;
216
217        // 采样段
218        let segment_count: usize = lines[cursor].parse().unwrap_or(0);
219        cursor += 1;
220
221        let mut segments = Vec::new();
222        for _ in 0..segment_count {
223            if cursor < lines.len() && !lines[cursor].is_empty() {
224                let parts: Vec<&str> = lines[cursor].split(',').collect();
225                if parts.len() >= 2 {
226                    let samp_rate = parts[0].trim().parse().unwrap_or(0.0);
227                    let end_point = parts[1].trim().parse().unwrap_or(0);
228                    segments.push(Segment::new(samp_rate, end_point));
229                }
230            }
231            cursor += 1;
232        }
233
234        let sampling = Sampling { freq, segments };
235
236        // 开始时间和故障时间
237        if cursor + 2 >= lines.len() {
238            return Err(Error::parse(FileRole::Cfg, 0, "时间行不足"));
239        }
240        let start_time = time::parse(lines[cursor])?;
241        cursor += 1;
242        let trigger_time = time::parse(lines[cursor])?;
243        cursor += 1;
244
245        // 数据格式
246        let data_type_text = lines[cursor].trim_end_matches(',').trim();
247        let data_type = DataType::parse(data_type_text).ok_or_else(|| {
248            Error::parse(
249                FileRole::Cfg,
250                cursor + 1,
251                format!("未知数据格式: {}", data_type_text),
252            )
253        })?;
254        cursor += 1;
255
256        // 可选字段
257        let mut timemult = 1.0;
258        let mut time_info = None;
259        let mut sampling_time_quality = None;
260
261        if cursor < lines.len() {
262            timemult = parse_f64_or(lines[cursor], 1.0);
263            cursor += 1;
264        }
265        if cursor < lines.len() {
266            let parts: Vec<&str> = lines[cursor].split(',').collect();
267            if parts.len() >= 2 {
268                time_info = Some(TimeInfo {
269                    time_code: parts[0].trim().to_string(),
270                    local_code: parts[1].trim().to_string(),
271                });
272            }
273            cursor += 1;
274        }
275        if cursor < lines.len() {
276            sampling_time_quality = Some(SamplingTimeQuality {
277                tmq_code: lines[cursor].trim().to_string(),
278            });
279        }
280
281        // 解析模拟量通道
282        let mut analogs = Vec::with_capacity(channels.analog);
283        for i in 0..channels.analog {
284            let line_idx = i + 2;
285            if line_idx < lines.len() {
286                let mut ch = AnalogChannel::from_cfg_line(lines[line_idx]);
287                if ch.index == 0 {
288                    ch.index = i + 1;
289                }
290                analogs.push(ch);
291            }
292        }
293
294        // 解析状态量通道
295        let mut statuses = Vec::with_capacity(channels.status);
296        let status_start = channels.analog + 2;
297        for i in 0..channels.status {
298            let line_idx = status_start + i;
299            if line_idx < lines.len() {
300                let mut ch = StatusChannel::from_cfg_line(lines[line_idx]);
301                if ch.index == 0 {
302                    ch.index = i + 1;
303                }
304                statuses.push(ch);
305            }
306        }
307
308        Ok(Config {
309            header,
310            channels,
311            analogs,
312            statuses,
313            sampling,
314            start_time,
315            trigger_time,
316            data_type,
317            timemult,
318            time_info,
319            sampling_time_quality,
320        })
321    }

323    /// 从文件读取 CFG 配置
324    pub fn from_file(path: &Path) -> Result<Config> {
325        let (text, _) = encoding::read_text_gbk(path)?;
326        Config::from_str(&text)
327    }

329    /// 序列化为 CFG 文本
330    #[allow(clippy::inherent_to_string)]
331    pub fn to_string(&self) -> String {
332        let mut lines = Vec::new();
333
334        // 头部
335        lines.push(self.header.to_line());
336        // 通道数
337        lines.push(self.channels.to_line());
338        // 模拟量通道
339        for analog in &self.analogs {
340            lines.push(analog.to_cfg_line());
341        }
342        // 状态量通道
343        for status in &self.statuses {
344            lines.push(status.to_cfg_line());
345        }
346        // 采样频率
347        lines.push(format!("{}", self.sampling.freq));
348        // 采样段数
349        lines.push(format!("{}", self.sampling.segments.len()));
350        for seg in &self.sampling.segments {
351            lines.push(format!("{:.0},{:.0}", seg.samp_rate, seg.end_point));
352        }
353        // 时间
354        lines.push(time::format_cfg(&self.start_time));
355        lines.push(time::format_cfg(&self.trigger_time));
356        // 数据格式
357        lines.push(self.data_type.as_str().to_string());
358        // timemult
359        lines.push(format!("{}", self.timemult));
360        // 可选字段
361        if let Some(ref ti) = self.time_info {
362            lines.push(format!("{},{}", ti.time_code, ti.local_code));
363        }
364        if let Some(ref stq) = self.sampling_time_quality {
365            lines.push(stq.tmq_code.clone());
366        }
367
368        lines.join("\n")
369    }

371    /// 写入 CFG 文件（默认 GBK 编码）
372    pub fn write_file(&self, path: &Path) -> Result<()> {
373        encoding::write_text(path, &self.to_string(), Encoding::Gbk)

387/// 从 "96A" / "192D" 类字符串中提取数字
388fn extract_digits(s: &str) -> std::result::Result<usize, std::num::ParseIntError> {
389    let digits: String = s.chars().filter(|c| c.is_ascii_digit()).collect();
390    if digits.is_empty() {
391        return Ok(0);
392    }
393    digits.parse()
394}
