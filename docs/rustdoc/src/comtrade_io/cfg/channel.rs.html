3/// 版本枚举
4///
5/// 六个版本号对应的标准（与 Python 基线 `model/type/version.py` 一致）：
6///
7/// | 版本 | 标准 |
8/// |---|---|
9/// | 1991 | IEEE Std C37.111-1991 |
10/// | 1999 | IEEE Std C37.111-1999 |
11/// | 2001 | IEC 60255-24:2001 |
12/// | 2008 | GB/T 22386-2008 |
13/// | 2013 | IEC 60255-24:2013 |
14/// | 2017 | GB/T 14598.24-2017 |
15///
16/// 1991 是唯一没有 CFG 可选尾部字段（`timemult` / 时间码 / 采样时间品质）的版本，
17/// 因此"是否 1999 及以后"是实际影响解析与序列化的判据，用 [`Version::is_1999_or_later`]。
18#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
19pub enum Version {
20    /// IEEE Std C37.111-1991
21    #[default]
22    V1991,
23    /// IEEE Std C37.111-1999
24    V1999,
25    /// IEC 60255-24:2001
26    V2001,
27    /// GB/T 22386-2008
28    V2008,
29    /// IEC 60255-24:2013
30    V2013,
31    /// GB/T 14598.24-2017
32    V2017,
33}
34
35impl Version {
36    /// 版本号字符串（如 `"1999"`）
37    pub fn as_str(&self) -> &'static str {
38        match self {
39            Version::V1991 => "1991",
40            Version::V1999 => "1999",
41            Version::V2001 => "2001",
42            Version::V2008 => "2008",
43            Version::V2013 => "2013",
44            Version::V2017 => "2017",
45        }
46    }
47
48    /// 该版本号对应的标准名称
49    pub fn standard(&self) -> &'static str {
50        match self {
51            Version::V1991 => "IEEE Std C37.111-1991",
52            Version::V1999 => "IEEE Std C37.111-1999",
53            Version::V2001 => "IEC 60255-24:2001",
54            Version::V2008 => "GB/T 22386-2008",
55            Version::V2013 => "IEC 60255-24:2013",
56            Version::V2017 => "GB/T 14598.24-2017",
57        }
58    }
59
60    /// 解析版本号字符串，无法识别时返回 `None`。
61    ///
62    /// 调用方通常用 `.unwrap_or_default()` 回落到 `V1991`——头部缺版本字段
63    /// 或版本号无法识别时按最早版本处理（对齐 Python `Version.from_value(x, V1991)`）。
64    pub fn parse(s: &str) -> Option<Version> {
65        match s.trim() {
66            "1991" => Some(Version::V1991),
67            "1999" => Some(Version::V1999),
68            "2001" => Some(Version::V2001),
69            "2008" => Some(Version::V2008),
70            "2013" => Some(Version::V2013),
71            "2017" => Some(Version::V2017),
72            _ => None,
73        }
74    }
75
76    /// 是否为 1999 及以后的版本。
77    ///
78    /// 1991 版 CFG 没有 `timemult` 及其后的可选行，1999 起才引入；
79    /// 2001/2008/2013/2017 均在 1999 的基础上扩展，可选字段布局与 1999 兼容。
80    /// 所有按版本分支的解析/序列化逻辑都应判断这个，而不是直接比 `== V1999`，
81    /// 否则 2001 及以后的文件会被当成 1991 处理。
82    pub fn is_1999_or_later(&self) -> bool {
83        !matches!(self, Version::V1991)
84    }
85}

87/// 数据格式枚举
88#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
89pub enum DataType {
90    /// ASCII 文本格式
91    #[default]
92    Ascii,
93    /// 二进制格式（16 位）
94    Binary,
95    /// 二进制格式（32 位）
96    Binary32,
97    /// 浮点格式（32 位，工程值直存）
98    Float32,
99}
100
101impl DataType {
102    /// 解析数据格式字符串（大小写不敏感），无法识别时返回 `None`
103    pub fn parse(s: &str) -> Option<DataType> {
104        match s.trim().to_uppercase().as_str() {
105            "ASCII" => Some(DataType::Ascii),
106            "BINARY" => Some(DataType::Binary),
107            "BINARY32" => Some(DataType::Binary32),
108            "FLOAT32" => Some(DataType::Float32),
109            _ => None,
110        }
111    }
112
113    /// 数据格式的 CFG 标准字符串（`ASCII` / `BINARY` / `BINARY32` / `FLOAT32`）
114    pub fn as_str(&self) -> &'static str {
115        match self {
116            DataType::Ascii => "ASCII",
117            DataType::Binary => "BINARY",
118            DataType::Binary32 => "BINARY32",
119            DataType::Float32 => "FLOAT32",
120        }
121    }
122
123    /// 模拟量是否 32 位存储
124    pub fn is_32bit(&self) -> bool {
125        matches!(self, DataType::Binary32 | DataType::Float32)
126    }
127}

129/// 转换标识
130#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
131pub enum TranSide {
132    /// 二次侧（Secondary）
133    #[default]
134    S,
135    /// 一次侧（Primary）
136    P,
137}
138
139impl TranSide {
140    /// 解析转换标识，`P` → 一次侧，其余（含缺省）→ 二次侧
141    pub fn parse(s: &str) -> TranSide {
142        match s.trim().to_uppercase().as_str() {
143            "P" => TranSide::P,
144            _ => TranSide::S,
145        }
146    }
147
148    /// 转换标识的 CFG 标准字符串（`P` / `S`）
149    pub fn as_str(&self) -> &'static str {
150        match self {
151            TranSide::P => "P",
152            TranSide::S => "S",
153        }
154    }
155}

176/// 模拟量通道模型
177#[derive(Debug, Clone)]
178pub struct AnalogChannel {
179    // —— CFG 原生 13 字段 ——
180    /// 通道编号（1 基）
181    pub index: usize,
182    /// 通道名（对应变电站一次设备的标识符）
183    pub name: String,
184    /// 相别标识（`A` / `B` / `C` / `N` 等）
185    pub phase: String,
186    /// 被监视的电路元件
187    pub equipment: String,
188    /// 单位（如 `V` / `A` / `W` / `Hz`）
189    pub unit: String,
190    /// 增益系数（`value = raw * multiplier + offset`）
191    pub multiplier: f64,
192    /// 偏移量
193    pub offset: f64,
194    /// 通道时滞（μs）
195    pub delay: f64,
196    /// 数值最小值（量程下限）
197    pub min_value: f64,
198    /// 数值最大值（量程上限）
199    pub max_value: f64,
200    /// 互感器一次系数
201    pub primary: f64,
202    /// 互感器二次系数
203    pub secondary: f64,
204    /// 转换标识（P/S）
205    pub tran_side: TranSide,
206    // —— INF/DMF 合并的扩展字段 ——
207    /// INF/DMF 合并的扩展字段（`None` 表示未提供）
208    pub ext: Option<AnalogExt>,
209}

211impl Default for AnalogChannel {
212    fn default() -> Self {
213        Self {
214            index: 0,
215            name: String::new(),
216            phase: String::new(),
217            equipment: String::new(),
218            unit: String::new(),
219            multiplier: 1.0,
220            offset: 0.0,
221            delay: 0.0,
222            min_value: 0.0,
223            max_value: 0.0,
224            primary: 1.0,
225            secondary: 1.0,
226            tran_side: TranSide::S,
227            ext: None,
228        }
229    }
230}
231
232impl AnalogChannel {
233    /// 从 CFG 行解析（13 字段，缺失尾部字段取默认值）
234    pub fn from_cfg_line(line: &str) -> AnalogChannel {
235        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
236        let index = parts.first().and_then(|s| s.parse().ok()).unwrap_or(0);
237        let name = parts.get(1).unwrap_or(&"").to_string();
238        let phase = parts.get(2).unwrap_or(&"").to_string();
239        let equipment = parts.get(3).unwrap_or(&"").to_string();
240        let unit = parts.get(4).unwrap_or(&"").to_string();
241        let multiplier = parse_f64_or(parts.get(5).unwrap_or(&""), 1.0);
242        let offset = parse_f64_or(parts.get(6).unwrap_or(&""), 0.0);
243        let delay = parse_f64_or(parts.get(7).unwrap_or(&""), 0.0);
244        let min_value = parse_f64_or(parts.get(8).unwrap_or(&""), 0.0);
245        let max_value = parse_f64_or(parts.get(9).unwrap_or(&""), 0.0);
246        let primary = parse_f64_or(parts.get(10).unwrap_or(&""), 1.0);
247        let secondary = parse_f64_or(parts.get(11).unwrap_or(&""), 1.0);
248        let tran_side = TranSide::parse(parts.get(12).unwrap_or(&"S"));
249
250        AnalogChannel {
251            index,
252            name,
253            phase,
254            equipment,
255            unit,
256            multiplier,
257            offset,
258            delay,
259            min_value,
260            max_value,
261            primary,
262            secondary,
263            tran_side,
264            ext: None,
265        }
266    }
267
268    /// 序列化为 CFG 行
269    pub fn to_cfg_line(&self) -> String {
270        format!(
271            "{},{},{},{},{},{},{},{},{},{},{},{},{}",
272            self.index,
273            self.name,
274            self.phase,
275            self.equipment,
276            self.unit,
277            fmt_float(self.multiplier),
278            fmt_float(self.offset),
279            fmt_float(self.delay),
280            fmt_float(self.min_value),
281            fmt_float(self.max_value),
282            fmt_float(self.primary),
283            fmt_float(self.secondary),
284            self.tran_side.as_str(),
285        )
286    }
287}

352/// 解析浮点数，非法返回默认值
353pub fn parse_f64_or(s: &str, default: f64) -> f64 {
354    if s.is_empty() {
355        return default;
356    }
357    s.parse().unwrap_or(default)
358}
359
360/// 浮点格式化：整值不显示小数点
361fn fmt_float(v: f64) -> String {
362    if v == v.trunc() && v.is_finite() {
363        format!("{:.0}", v)
364    } else {
365        format!("{}", v)
366    }
