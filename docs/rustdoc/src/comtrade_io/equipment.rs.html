9/// 设备组
10#[derive(Debug, Clone, Default)]
11pub struct EquipmentGroup {
12    /// 母线列表

24/// 交流电压通道组（DMF `ACVChn` / INF `TV_CHNS`）
25///
26/// 通道号为 1 基（对应 CFG 通道编号），`0` 表示未配置。
27#[derive(Debug, Clone, Default, PartialEq)]
28pub struct AcvChn {
29    /// A 相电压通道号（对应 CFG 模拟量通道编号）
30    pub ua_idx: usize,
31    /// B 相电压通道号
32    pub ub_idx: usize,
33    /// C 相电压通道号
34    pub uc_idx: usize,
35    /// N 相（零序）电压通道号
36    pub un_idx: usize,
37    /// L 相（线电压）通道号
38    pub ul_idx: usize,
39}
40
41impl AcvChn {
42    /// 是否全部通道均未配置
43    pub fn is_empty(&self) -> bool {
44        self.ua_idx == 0
45            && self.ub_idx == 0
46            && self.uc_idx == 0
47            && self.un_idx == 0
48            && self.ul_idx == 0
49    }
50}

52/// 交流电流分支（DMF `ACC_Bran` / INF `TA_CHNS` `TA_Id_#N`）
53///
54/// `dir` 语义按规范 §1.4：`1` = 正方向（流向线路），`-1` = 反方向，缺省 `1`。
55/// 变压器侧（§1.5）该值为极性符号：`1` = 电流流入变压器，`-1` = 流出。
56///
57/// `dir_raw` 保留 DMF 原始拼写（样本中 `POS` / `pos` 大小写不一致），
58/// 以便 round-trip 时按原文写回；`dir` 是归一化后的语义值。
59#[derive(Debug, Clone, PartialEq)]
60pub struct AccBran {
61    /// 分支序号（DMF `bran_idx`，从 1 起）
62    pub bran_idx: usize,
63    /// A 相电流通道号
64    pub ia_idx: usize,
65    /// B 相电流通道号
66    pub ib_idx: usize,
67    /// C 相电流通道号
68    pub ic_idx: usize,
69    /// N 相（零序）电流通道号
70    pub in_idx: usize,
71    /// 电流方向语义值（`1` = 正方向，`-1` = 反方向）
72    pub dir: i32,
73    /// 方向原始拼写（DMF `dir` 属性原文，如 `POS`/`pos`）
74    pub dir_raw: String,
75}
76
77impl Default for AccBran {
78    fn default() -> Self {
79        AccBran {
80            bran_idx: 0,
81            ia_idx: 0,
82            ib_idx: 0,
83            ic_idx: 0,
84            in_idx: 0,
85            dir: 1,
86            dir_raw: String::new(),
87        }
88    }
89}
90
91impl AccBran {
92    /// 是否全部通道均未配置
93    pub fn is_empty(&self) -> bool {
94        self.ia_idx == 0 && self.ib_idx == 0 && self.ic_idx == 0 && self.in_idx == 0
95    }
96}

277/// 变压器绕组位置
278#[derive(Debug, Clone, Copy, PartialEq, Eq)]
279pub enum WindingLocation {
280    /// 高压侧
281    High,
282    /// 中压侧
283    Medium,
284    /// 低压侧
285    Low,
286}
287
288impl WindingLocation {
289    /// INF 段 Key 前缀（`H_` / `M_` / `L_`）
290    pub fn inf_prefix(&self) -> &'static str {
291        match self {
292            WindingLocation::High => "H",
293            WindingLocation::Medium => "M",
294            WindingLocation::Low => "L",
295        }
296    }
297
298    /// 该侧的 `TA_Id_#N` 分支序号。
299    ///
300    /// 依据规范 §1.5：`#1/#2` = 高压侧一/二分支，`#3/#4` = 中压侧一/二分支，
301    /// `#5/#6/#7` = 低压侧一/二/三分支。
302    pub fn ta_id_indices(&self) -> &'static [usize] {
303        match self {
304            WindingLocation::High => &[1, 2],
305            WindingLocation::Medium => &[3, 4],
306            WindingLocation::Low => &[5, 6, 7],
307        }
308    }
309
310    /// DMF `location` 属性的规范拼写
311    pub fn as_str(&self) -> &'static str {
312        match self {
313            WindingLocation::High => "High",
314            WindingLocation::Medium => "Medium",
315            WindingLocation::Low => "Low",
316        }
317    }
318
319    /// 从 DMF `location` 属性解析（大小写不敏感）
320    pub fn parse(s: &str) -> Option<WindingLocation> {
321        match s.trim().to_ascii_lowercase().as_str() {
322            "high" | "h" => Some(WindingLocation::High),
323            "medium" | "middle" | "m" => Some(WindingLocation::Medium),
324            "low" | "l" => Some(WindingLocation::Low),
325            _ => None,
326        }
327    }
328}
