1//! 编码探测与转码模块
2//!
3//! 支持 UTF-8（含 BOM 剥离）、GBK、cp1251 编码。
4
5use std::path::Path;
6
7use crate::error::{Error, Result};
8
9/// 文本编码类型
10#[derive(Debug, Clone, Copy, PartialEq, Eq)]
11pub enum Encoding {
12    /// UTF-8（含 BOM 剥离）
13    Utf8,
14    /// GBK（中文编码）
15    Gbk,
16    /// cp1251（西里尔，DFR WNDR 头专用）
17    Cp1251,
18}
19
20/// 探测字节序列的编码（UTF-8 严格校验优先，失败回退 GBK）
21pub fn detect(bytes: &[u8]) -> Encoding {
22    // 剥离 BOM
23    let data = strip_bom(bytes);
24    if std::str::from_utf8(data).is_ok() {
25        Encoding::Utf8
26    } else {
27        Encoding::Gbk
28    }
29}
30
31/// 剥离 UTF-8 BOM（如果存在）
32fn strip_bom(bytes: &[u8]) -> &[u8] {
33    if bytes.len() >= 3 && bytes[0] == 0xEF && bytes[1] == 0xBB && bytes[2] == 0xBF {
34        &bytes[3..]
35    } else {
36        bytes
37    }
38}
39
40/// 自动探测并解码为 String
41pub fn decode(bytes: &[u8]) -> String {
42    let enc = detect(bytes);
43    decode_with(bytes, enc)
44}
45
46/// 按指定编码解码
47pub fn decode_with(bytes: &[u8], enc: Encoding) -> String {
48    match enc {
49        Encoding::Utf8 => {
50            let data = strip_bom(bytes);
51            String::from_utf8_lossy(data).into_owned()
52        },
53        Encoding::Gbk => decode_gbk(bytes),
54        Encoding::Cp1251 => decode_cp1251(bytes),
55    }
56}
57
58/// 编码为字节
59pub fn encode(text: &str, enc: Encoding) -> Vec<u8> {
60    match enc {
61        Encoding::Utf8 => text.as_bytes().to_vec(),
62        Encoding::Gbk => encode_gbk(text),
63        Encoding::Cp1251 => {
64            // cp1251 不支持编码回写（DFR 只读）
65            text.as_bytes().to_vec()
66        },
67    }
68}
69
70/// 读取文本文件（自动探测编码）
71#[allow(dead_code)]
72pub fn read_text(path: &Path) -> Result<(String, Encoding)> {
73    let bytes = std::fs::read(path).map_err(|e| Error::Io {
74        path: path.to_path_buf(),
75        source: e,
76    })?;
77    let enc = detect(&bytes);
78    let text = decode_with(&bytes, enc);
79    Ok((text, enc))
80}
81
82/// 读取文本文件，按 GBK 优先（对齐 Python CFG/INF 读取行为）
83pub fn read_text_gbk(path: &Path) -> Result<(String, Encoding)> {
84    let bytes = std::fs::read(path).map_err(|e| Error::Io {
85        path: path.to_path_buf(),
86        source: e,
87    })?;
88    // GBK 优先，失败回退 UTF-8
89    let text = decode_gbk(&bytes);
90    if text.contains('\u{FFFD}') {
91        // 含替换符，尝试 UTF-8
92        match std::str::from_utf8(&bytes) {
93            Ok(s) => Ok((s.to_string(), Encoding::Utf8)),
94            Err(_) => Ok((text, Encoding::Gbk)),
95        }
96    } else {
97        Ok((text, Encoding::Gbk))
98    }
99}

101/// 写出文本文件（指定编码）
102pub fn write_text(path: &Path, text: &str, enc: Encoding) -> Result<()> {
103    let bytes = encode(text, enc);
104    std::fs::write(path, bytes).map_err(|e| Error::Io {
105        path: path.to_path_buf(),
106        source: e,
107    })
108}

110// ========== GBK 编解码（使用 encoding_rs） ==========
111
112#[cfg(not(feature = "gbk-builtin"))]
113fn decode_gbk(bytes: &[u8]) -> String {
114    use encoding_rs::GBK;
115    GBK.decode_without_bom_handling(bytes).0.into_owned()
116}
117
118#[cfg(not(feature = "gbk-builtin"))]
119fn encode_gbk(text: &str) -> Vec<u8> {
120    use encoding_rs::GBK;
121    GBK.encode(text).0.into_owned()
122}
123
124#[cfg(feature = "gbk-builtin")]
125fn decode_gbk(bytes: &[u8]) -> String {
126    gbk_builtin::decode(bytes)
127}
128
129#[cfg(feature = "gbk-builtin")]
130fn encode_gbk(text: &str) -> Vec<u8> {
131    gbk_builtin::encode(text)
132}

134#[cfg(feature = "gbk-builtin")]
135mod gbk_builtin {
136    /// 内置 GBK 解码（精简实现，未映射字符用 U+FFFD）
137    pub fn decode(bytes: &[u8]) -> String {
138        let mut result = String::with_capacity(bytes.len());
139        let mut i = 0;
140        while i < bytes.len() {
141            let b = bytes[i];
142            if b < 0x80 {
143                result.push(b as char);
144                i += 1;
145            } else if b >= 0x81 && b <= 0xFE && i + 1 < bytes.len() {
146                let b2 = bytes[i + 1];
147                if (0x40..=0x7E).contains(&b2) || (0x80..=0xFE).contains(&b2) {
148                    let code = ((b as u32) << 8) | (b2 as u32);
149                    result.push(gbk_to_unicode(code));
150                    i += 2;
151                } else {
152                    result.push('\u{FFFD}');
153                    i += 1;
154                }
155            } else {
156                result.push('\u{FFFD}');
157                i += 1;
158            }
159        }
160        result
161    }
162
163    pub fn encode(_text: &str) -> Vec<u8> {
164        // 精简版：仅 ASCII 区间直通，其余返回原始字节
165        // 完整版需要 GBK 码表，约 2.3 万条目
166        let mut result = Vec::new();
167        for ch in _text.chars() {
168            let c = ch as u32;
169            if c < 0x80 {
170                result.push(c as u8);
171            } else if let Some((hi, lo)) = unicode_to_gbk(c) {
172                result.push(hi);
173                result.push(lo);
174            } else {
175                result.push(b'?');
176            }
177        }
178        result
179    }
180
181    fn gbk_to_unicode(code: u32) -> char {
182        // 精简映射表：仅覆盖常用 CJK 区间
183        match code {
184            // GBK/2 区：GB2312 汉字
185            0xB0A1..=0xD7F9 => {
186                // 简化实现：U+FFFD 表示未映射
187                // 实际需要完整码表
188                '\u{FFFD}'
189            },
190            0xD8A1..=0xF7FE => '\u{FFFD}',
191            // GBK/3 区
192            0x8140..=0xA0FE => '\u{FFFD}',
193            // GBK/4 区
194            0xAA40..=0xFEA0 => '\u{FFFD}',
195            _ => '\u{FFFD}',
196        }
197    }
198
199    fn unicode_to_gbk(_cp: u32) -> Option<(u8, u8)> {
200        None
201    }
202}

226fn decode_cp1251(bytes: &[u8]) -> String {
227    bytes
228        .iter()
229        .map(|&b| {
230            if b < 0x80 {
231                b as char
232            } else {
233                CP1251_TABLE[(b - 0x80) as usize]
234            }
235        })
236        .collect()
237}
