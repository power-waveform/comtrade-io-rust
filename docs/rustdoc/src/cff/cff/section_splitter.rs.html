1//! CFF 字节级段切分
2//!
3//! 在字节流上定位 `--- file type: XXX ---` 标记，段文本按 UTF-8 优先、
4//! 失败回退 GBK 的策略解码（与 CFG/INF 的 `read_text_gbk` 行为对齐）。
5//! BINARY/FLOAT32 DAT 段保留原始字节，不做文本解码也不做空白裁剪。
6
7/// CFF 段内容
8#[derive(Debug, Clone, Default)]
9pub struct CffSections {
10    /// CFG 段文本
11    pub cfg: Option<String>,
12    /// DAT 段文本（ASCII 格式）
13    pub dat: Option<String>,
14    /// DAT 段原始字节（BINARY/FLOAT32 格式）
15    pub dat_bytes: Option<Vec<u8>>,
16    /// INF 段文本
17    pub inf: Option<String>,
18    /// HDR 段文本
19    pub hdr: Option<String>,
20}
21
22/// 提取 CFF 段
23pub fn extract_sections(data: &[u8]) -> CffSections {
24    let mut result = CffSections::default();
25    let markers = find_markers(data);
26
27    for &(section_type, start, end) in &markers {
28        let section_bytes = &data[start..end];
29
30        match section_type {
31            "CFG" => {
32                result.cfg = Some(decode_text(trim_bytes(section_bytes)));
33            },
34            "DAT" => {
35                // 文本（ASCII）DAT 段：解码后供 `DatFile::from_ascii` 使用
36                result.dat = Some(decode_text(trim_bytes(section_bytes)));
37                // 二进制 DAT 段：保留**未裁剪**的原始字节。
38                // BINARY/FLOAT32 采样字节合法地等于 0x20/0x09/0x0A/0x0D，
39                // 两端裁剪会静默吃掉首尾样本并移位后续全部数据。
40                result.dat_bytes = Some(section_bytes.to_vec());
41            },
42            "INF" => {
43                result.inf = Some(decode_text(trim_bytes(section_bytes)));
44            },
45            "HDR" => {
46                result.hdr = Some(decode_text(trim_bytes(section_bytes)));
47            },
48            _ => {},
49        }
50    }
51
52    result
53}

55/// 查找所有段标记，返回 (类型, 内容开始, 内容结束)
56fn find_markers(data: &[u8]) -> Vec<(&str, usize, usize)> {
57    let mut markers = Vec::new();
58    let mut pos = 0;
59
60    while pos < data.len() {
61        // 查找行首的 "--" 或 "---"
62        let is_start = pos == 0 || data[pos - 1] == b'\n';
63
64        if is_start && pos + 1 < data.len() && data[pos] == b'-' && data[pos + 1] == b'-' {
65            // 查找这一行的结束
66            let mut line_end = pos;
67            while line_end < data.len() && data[line_end] != b'\n' && data[line_end] != b'\r' {
68                line_end += 1;
69            }
70            let line = &data[pos..line_end];
71
72            // 尝试匹配 "--- file type: XXX ---" 模式
73            if let Some(section_type) = match_section_line(line) {
74                // 跳过行尾换行符
75                let content_start = skip_newlines(data, line_end);
76                markers.push((section_type, content_start, data.len())); // 先暂存 end
77                if markers.len() > 1 {
78                    // 更新前一个标记的 end
79                    let prev_end = pos;
80                    let prev_idx = markers.len() - 2;
81                    markers[prev_idx].2 = prev_end;
82                }
83            }
84        }
85
86        pos += 1;
87    }
88
89    // 最后一个标记的 end 已经是 data.len()
90    markers
91}

93/// 匹配段标记行，返回段类型
94fn match_section_line(line: &[u8]) -> Option<&str> {
95    let text = std::str::from_utf8(line).ok()?;
96    let lower = text.trim().to_lowercase();
97
98    // 必须以 -- 或 --- 开头，以 --- 结尾
99    if !lower.starts_with("--") || !lower.ends_with("---") {
100        return None;
101    }
102
103    // 去掉开头的 -- 或 ---
104    let inner = if let Some(stripped) = lower.strip_prefix("---") {
105        stripped
106    } else {
107        lower.strip_prefix("--").unwrap_or(&lower)
108    };
109    let inner = inner[..inner.len() - 3].trim();
110    if !inner.contains("file") || !inner.contains("type") {
111        return None;
112    }
113
114    // 提取段类型
115    for part in inner.split_whitespace() {
116        let cleaned = part.trim_end_matches(':');
117        match cleaned.to_uppercase().as_str() {
118            "CFG" => return Some("CFG"),
119            "DAT" => return Some("DAT"),
120            "INF" => return Some("INF"),
121            "HDR" => return Some("HDR"),
122            _ => {},
123        }
124    }
125
126    // 在 "file type" 后面找类型
127    if let Some(type_pos) = inner.find("type") {
128        let after = &inner[type_pos + 4..].trim();
129        let after = after.trim_start_matches(':').trim();
130        let word = after.split_whitespace().next().unwrap_or("");
131        match word.to_uppercase().as_str() {
132            "CFG" => return Some("CFG"),
133            "DAT" => return Some("DAT"),
134            "INF" => return Some("INF"),
135            "HDR" => return Some("HDR"),
136            _ => {},
137        }
138    }
139
140    None
141}

143fn skip_newlines(data: &[u8], pos: usize) -> usize {
144    let mut p = pos;
145    while p < data.len() && (data[p] == b'\n' || data[p] == b'\r') {
146        p += 1;
147    }
148    p
149}

151fn trim_bytes(data: &[u8]) -> &[u8] {
152    let mut start = 0;
153    let mut end = data.len();
154    while start < end && data[start].is_ascii_whitespace() {
155        start += 1;
156    }
157    while end > start && data[end - 1].is_ascii_whitespace() {
158        end -= 1;
159    }
160    &data[start..end]
161}

163/// 按段独立探测编码并解码：UTF-8 严格校验优先，失败回退 GBK。
164///
165/// 必须**按段**探测而非对整个文件一次性判定——BINARY DAT 段永远不是合法
166/// UTF-8，若参与全局判定会把 CFG 段的 UTF-8 中文拖成 GBK 误读。
167fn decode_text(data: &[u8]) -> String {
168    cbase::encoding::decode(data)
169}

171#[cfg(test)]
172mod tests {
173    use super::*;

175    #[test]
176    fn test_extract_cff_sections() {
177        let data = b"--- file type CFG ---\nSTATION,RECORDER,1999\n...\n--- file type DAT ---\n1,0,100,200\n--- file type INF ---\n[Public File_Description]\nStation_Name=TEST\n";
178        let sections = extract_sections(data);
179        assert!(sections.cfg.is_some());
180        assert!(sections.dat.is_some());
181        assert!(sections.inf.is_some());
182    }
183
184    #[test]
185    fn test_dat_bytes_not_trimmed() {
186        // BINARY DAT 段首尾字节合法地等于 ASCII 空白值（0x20/0x09/0x0A/0x0D），
187        // 绝不能被 trim_bytes 裁掉——否则首尾样本丢失、后续样本移位。
188        let payload: &[u8] = &[0x20, 0x0A, 0x00, 0x01, 0x09, 0x0D, 0x20, 0x20];
189        let mut data = Vec::new();
190        data.extend_from_slice(b"--- file type CFG ---\nSTATION,X,1999\n...\n");
191        data.extend_from_slice(b"--- file type DAT ---\n");
192        data.extend_from_slice(payload);
193        let sections = extract_sections(&data);
194        let dat_bytes = sections.dat_bytes.expect("应有 DAT 字节段");
195        assert_eq!(dat_bytes, payload, "BINARY DAT 字节必须逐字保留，不得裁剪");
196    }
197
198    #[test]
199    fn test_utf8_cfg_section_not_misread_as_gbk() {
200        // UTF-8 编码的中文不应被 GBK 解码成乱码。
201        // 「母线」= E6 AF 8D E7 BA BF（UTF-8）。
202        let cfg = "--- file type CFG ---\nGHBZ,220kV\u{6bcd}\u{7ebf}\u{6545}\u{969c},2013\n1,1,1,1,50,1,1,1,50A,1,1,0,0,0,0,0\n0\n";
203        let data = cfg.as_bytes();
204        let sections = extract_sections(data);
205        let cfg_text = sections.cfg.expect("应有 CFG 段");
206        assert!(
207            cfg_text.contains('\u{6bcd}'),
208            "UTF-8 中文应正确解码，实际: {}",
209            cfg_text
210        );
211    }
