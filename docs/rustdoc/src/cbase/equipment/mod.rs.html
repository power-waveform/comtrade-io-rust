1//! 设备模型（INF/DMF 共享）
2//!
3//! Bus/Line/Transformer 设备拓扑模型，被 DMF 和 INF 模块复用。
4//!
5//! 字段命名以 DMF XML 属性与 INF 段 Key 为准，规范依据：
6//! - DMF：`docs/rust重构需求分析报告.md` §2.6
7//! - INF：`docs/COMTRADE1999补充定义(090917).pdf` §1.4（线路段）/ §1.5（变压器段）
8
9pub mod bus;
10pub mod generator;
11pub mod line;
12pub mod transformer;
13
14/// 设备组
15#[derive(Debug, Clone, Default)]
16pub struct EquipmentGroup {
17    /// 母线列表
18    pub buses: Vec<Bus>,
19    /// 线路列表
20    pub lines: Vec<Line>,
21    /// 变压器列表
22    pub transformers: Vec<Transformer>,
23    /// 发电机（规范 §1.6 `[ZYHD POWER_#n]`）
24    pub generators: Vec<Generator>,
25    /// 励磁机（规范 §1.7 `[ZYHD EXCITATION_#n]`）
26    pub exciters: Vec<Exciter>,
27}
28
29/// 交流电压通道组（DMF `ACVChn` / INF `TV_CHNS`）
30///
31/// 通道号为 1 基（对应 CFG 通道编号），`0` 表示未配置。
32#[derive(Debug, Clone, Default, PartialEq)]
33pub struct AcvChn {
34    /// A 相电压通道号（对应 CFG 模拟量通道编号）
35    pub ua_idx: usize,
36    /// B 相电压通道号
37    pub ub_idx: usize,
38    /// C 相电压通道号
39    pub uc_idx: usize,
40    /// N 相（零序）电压通道号
41    pub un_idx: usize,
42    /// L 相（线电压）通道号
43    pub ul_idx: usize,
44}
45
46impl AcvChn {
47    /// 是否全部通道均未配置
48    pub fn is_empty(&self) -> bool {
49        self.ua_idx == 0
50            && self.ub_idx == 0
51            && self.uc_idx == 0
52            && self.un_idx == 0
53            && self.ul_idx == 0
54    }
55}
56
57/// 交流电流分支（DMF `ACC_Bran` / INF `TA_CHNS` `TA_Id_#N`）
58///
59/// `dir` 语义按规范 §1.4：`1` = 正方向（流向线路），`-1` = 反方向，缺省 `1`。
60/// 变压器侧（§1.5）该值为极性符号：`1` = 电流流入变压器，`-1` = 流出。
61///
62/// `dir_raw` 保留 DMF 原始拼写（样本中 `POS` / `pos` 大小写不一致），
63/// 以便 round-trip 时按原文写回；`dir` 是归一化后的语义值。
64#[derive(Debug, Clone, PartialEq)]
65pub struct AccBran {
66    /// 分支序号（DMF `bran_idx`，从 1 起）
67    pub bran_idx: usize,
68    /// A 相电流通道号
69    pub ia_idx: usize,
70    /// B 相电流通道号
71    pub ib_idx: usize,
72    /// C 相电流通道号
73    pub ic_idx: usize,
74    /// N 相（零序）电流通道号
75    pub in_idx: usize,
76    /// 电流方向语义值（`1` = 正方向，`-1` = 反方向）
77    pub dir: i32,
78    /// 方向原始拼写（DMF `dir` 属性原文，如 `POS`/`pos`）
79    pub dir_raw: String,
80}
81
82impl Default for AccBran {
83    fn default() -> Self {
84        AccBran {
85            bran_idx: 0,
86            ia_idx: 0,
87            ib_idx: 0,
88            ic_idx: 0,
89            in_idx: 0,
90            dir: 1,
91            dir_raw: String::new(),
92        }
93    }
94}
95
96impl AccBran {
97    /// 是否全部通道均未配置
98    pub fn is_empty(&self) -> bool {
99        self.ia_idx == 0 && self.ib_idx == 0 && self.ic_idx == 0 && self.in_idx == 0
100    }
101}
