11/// 构建设备组
12///
13/// 偏离 Python 基线：原 Rust 实现读 `bus_name` / `line_name` / `trm_name`
14/// （DMF XML 的属性名），而 INF 段用的是 `DEV_ID=idx,name`，
15/// 结果全部字段落空——设备只剩空名字和节序号。此处按 INF 真实 Key 解析。
16pub fn build_equipment_group(inf: &InfFile) -> EquipmentGroup {
17    let mut eg = EquipmentGroup::default();
18
19    for section in &inf.sections {
20        match &section.kind {
21            SectionKind::Bus => {
22                let (idx, name) = split_dev_id(section.get("DEV_ID").unwrap_or(""), section.index);
23                eg.buses.push(Bus {
24                    idx,
25                    name,
26                    src_ref: String::new(),
27                    sys_id: section.get("SYS_ID").unwrap_or("").to_string(),
28                    // `RATED_VALUE=220000V` 是一次额定电压，单位 V；模型以 kV 存储
29                    v_rtg: section
30                        .get("RATED_VALUE")
31                        .and_then(parse_num_with_unit)
32                        .map(|v| v / 1000.0)
33                        .unwrap_or(0.0),
34                    v_rtg_snd: 0.0,
35                    tv_pos: section.get("TV_POS").unwrap_or("").to_string(),
36                    is_location: String::new(),
37                    bus_uuid: String::new(),
38                    acv: section.get("TV_CHNS").map(parse_acv).unwrap_or_default(),
39                    ana_chns: section
40                        .get("OTH_ACHNS")
41                        .map(parse_num_list)
42                        .unwrap_or_default(),
43                    sta_chns: section
44                        .get("STATUS_CHNS")
45                        .map(parse_num_list)
46                        .unwrap_or_default(),
47                });
48            },
49            SectionKind::Line => {
50                let (idx, name) = split_dev_id(section.get("DEV_ID").unwrap_or(""), section.index);
51
52                // 电流分支：`TA_CHNS` 为第一组，`TA_CHNS_#2` 为第二组
53                let mut currents = Vec::new();
54                if let Some(raw) = section.get("TA_CHNS") {
55                    currents.push(parse_line_ta(raw, 1));
56                }
57                if let Some(raw) = section.get("TA_CHNS_#2") {
58                    currents.push(parse_line_ta(raw, 2));
59                }
60
61                // 电压通道组：`TV_CHNS` 与 `TV_CHNS_#n` 兼容并存（规范 §1.4）
62                let mut voltages = Vec::new();
63                if let Some(raw) = section.get("TV_CHNS") {
64                    voltages.push(parse_acv(raw));
65                }
66                for n in 1..=8 {
67                    if let Some(raw) = section.get(&format!("TV_CHNS_#{}", n)) {
68                        let acv = parse_acv(raw);
69                        if !voltages.contains(&acv) {
70                            voltages.push(acv);
71                        }
72                    }
73                }
74
75                let bran_num = currents.len();
76                eg.lines.push(Line {
77                    idx,
78                    name,
79                    bus_id: String::new(),
80                    src_ref: String::new(),
81                    sys_id: section.get("SYS_ID").unwrap_or("").to_string(),
82                    object_type: section.get("OBJECT_TYPE").unwrap_or("").to_string(),
83                    v_rtg: 0.0,
84                    a_rtg: 0.0,
85                    a_rtg_snd: 0.0,
86                    line_len: section
87                        .get("LENGTH")
88                        .and_then(parse_num_with_unit)
89                        .unwrap_or(0.0),
90                    bran_num,
91                    line_uuid: String::new(),
92                    remote_id: String::new(),
93                    remote_flag: String::new(),
94                    differential_id: String::new(),
95                    other_id: section.get("OTHER_ID").unwrap_or("").to_string(),
96                    reactor: section.get("REACTOR").and_then(parse_reactor),
97                    impedance: section.get("RX").map(parse_impedance).unwrap_or_default(),
98                    capacitance: section.get("CG").map(parse_capacitance).unwrap_or_default(),
99                    mutual_inductance: section.get("MRX").map(parse_mutual).unwrap_or_default(),
100                    px: Px::default(),
101                    currents,
102                    voltages,
103                    differential_current: AcvChn::default(),
104                    oth_achns: section
105                        .get("OTH_ACHNS")
106                        .map(parse_num_list)
107                        .unwrap_or_default(),
108                    ana_chns: Vec::new(),
109                    sta_chns: section
110                        .get("STATUS_CHNS")
111                        .map(parse_num_list)
112                        .unwrap_or_default(),
113                });
114            },
115            SectionKind::Transformer => {
116                let (idx, name) = split_dev_id(section.get("DEV_ID").unwrap_or(""), section.index);
117                let winding_num = section
118                    .get("WINDING_NUM")
119                    .and_then(parse_num_with_unit)
120                    .map(|v| v.max(0.0) as usize)
121                    .unwrap_or(0);
122
123                let windings = winding_layout(winding_num)
124                    .iter()
125                    .filter_map(|loc| build_winding(section, *loc))
126                    .collect();
127
128                eg.transformers.push(Transformer {
129                    idx,
130                    name,
131                    src_ref: String::new(),
132                    sys_id: section.get("SYS_ID").unwrap_or("").to_string(),
133                    object_type: section.get("OBJECT_TYPE").unwrap_or("").to_string(),
134                    pwr_rtg: section
135                        .get("CAPACITY")
136                        .and_then(parse_num_with_unit)
137                        .unwrap_or(0.0),
138                    winding_num,
139                    ta_self_comp: section.get("TA_SELF_COMP").unwrap_or("").to_string(),
140                    transformer_uuid: String::new(),
141                    windings,
142                    oth_achns: section
143                        .get("OTH_ACHNS")
144                        .map(parse_num_list)
145                        .unwrap_or_default(),
146                    ana_chns: Vec::new(),
147                    sta_chns: section
148                        .get("STATUS_CHNS")
149                        .map(parse_num_list)
150                        .unwrap_or_default(),
151                });
152            },
153            SectionKind::Generator => {
154                // 规范 §1.6 `[ZYHD POWER_#n]`
155                let (idx, name) = split_dev_id(section.get("DEV_ID").unwrap_or(""), section.index);
156                eg.generators.push(Generator {
157                    idx,
158                    name,
159                    src_ref: String::new(),
160                    sys_id: section.get("SYS_ID").unwrap_or("").to_string(),
161                    // TRM_ID 与 DEV_ID 同语法（idx,name），原文存储保 round-trip
162                    trm_id: section.get("TRM_ID").unwrap_or("").to_string(),
163                    object_type: section.get("OBJECT_TYPE").unwrap_or("").to_string(),
164                    freq: section
165                        .get("FREQ")
166                        .and_then(parse_num_with_unit)
167                        .unwrap_or(0.0),
168                    capacity: section
169                        .get("CAPACITY")
170                        .and_then(parse_num_with_unit)
171                        .unwrap_or(0.0),
172                    factor: section
173                        .get("FACTOR")
174                        .and_then(parse_num_with_unit)
175                        .unwrap_or(0.0),
176                    v1: section
177                        .get("V1")
178                        .and_then(parse_num_with_unit)
179                        .unwrap_or(0.0),
180                    branch_num: parse_branch_num(section.get("BRANCH_NUM")),
181                    rotor_i: section
182                        .get("Rotor_I")
183                        .and_then(parse_num_with_unit)
184                        .unwrap_or(0.0),
185                    rotor_v2: section
186                        .get("Rotor_V2")
187                        .and_then(parse_num_with_unit)
188                        .unwrap_or(0.0),
189                    ufe: parse_ufe(section.get("Ufe")),
190                    x: parse_sync_reactance(section.get("X")),
191                    excitation_mode: section
192                        .get("EXCITATION_MODE")
193                        .and_then(parse_num_with_unit)
194                        .map(|v| v as i32)
195                        .unwrap_or(0),
196                    igt_dir: section
197                        .get("IGT_DIR")
198                        .and_then(parse_num_with_unit)
199                        .map(|v| v as i32)
200                        .unwrap_or(0),
201                    // TV_CHNS 仅 3 字段（Ua,Ub,Uc），un_idx 恒 0
202                    acv: parse_acv(section.get("TV_CHNS").unwrap_or("")),
203                    ta: parse_transformer_ta(section.get("TA_CHNS").unwrap_or(""), 1),
204                    ta_z1: parse_transformer_ta(section.get("TA_Z1_CHNS").unwrap_or(""), 1),
205                    ta_z2: parse_transformer_ta(section.get("TA_Z2_CHNS").unwrap_or(""), 2),
206                    ta_z3: parse_transformer_ta(section.get("TA_Z3_CHNS").unwrap_or(""), 3),
207                    ufe_chns: parse_ufe_chns(section.get("Ufe_CHNS")),
208                    ife_chn: section
209                        .get("Ife_CHN")
210                        .and_then(parse_num_with_unit)
211                        .map(|v| v.max(0.0) as usize)
212                        .unwrap_or(0),
213                    un_chns: parse_un_chns(section.get("UN_CHNS")),
214                    ta_ido_chn: section
215                        .get("TA_Ido_CHN")
216                        .and_then(parse_num_with_unit)
217                        .map(|v| v.max(0.0) as usize)
218                        .unwrap_or(0),
219                    oth_achns: section
220                        .get("OTH_ACHNS")
221                        .map(parse_num_list)
222                        .unwrap_or_default(),
223                    sta_chns: section
224                        .get("STATUS_CHNS")
225                        .map(parse_num_list)
226                        .unwrap_or_default(),
227                });
228            },
229            SectionKind::Exciter => {
230                // 规范 §1.7 `[ZYHD EXCITATION_#n]`
231                let (idx, name) = split_dev_id(section.get("DEV_ID").unwrap_or(""), section.index);
232                eg.exciters.push(Exciter {
233                    idx,
234                    name,
235                    src_ref: String::new(),
236                    sys_id: section.get("SYS_ID").unwrap_or("").to_string(),
237                    // PWR_ID 与 DEV_ID 同语法（idx,name），原文存储保 round-trip
238                    pwr_id: section.get("PWR_ID").unwrap_or("").to_string(),
239                    object_type: section.get("OBJECT_TYPE").unwrap_or("").to_string(),
240                    freq: section
241                        .get("FREQ")
242                        .and_then(parse_num_with_unit)
243                        .unwrap_or(0.0),
244                    v1: section
245                        .get("V1")
246                        .and_then(parse_num_with_unit)
247                        .unwrap_or(0.0),
248                    // TV_CHNS 为 4 字段（Ua,Ub,Uc,Un）
249                    acv: parse_acv(section.get("TV_CHNS").unwrap_or("")),
250                    ta: parse_transformer_ta(section.get("TA_CHNS").unwrap_or(""), 1),
251                    ta_z: parse_transformer_ta(section.get("TA_Z_CHNS").unwrap_or(""), 1),
252                    oth_achns: section
253                        .get("OTH_ACHNS")
254                        .map(parse_num_list)
255                        .unwrap_or_default(),
256                    sta_chns: section
257                        .get("STATUS_CHNS")
258                        .map(parse_num_list)
259                        .unwrap_or_default(),
260                });
261            },
262            _ => {},
263        }
264    }
265
266    eg

271/// 解析带单位后缀的数值，如 `22.45(km)` / `220000V` / `0.071(Ω/km)` / `180(MVA)`。
272///
273/// 保留前导负号（`REACTOR=-1(Ω)` 是有效值），从首个非数值字符处截断。
274/// 无法解析时返回 `None`。
275fn parse_num_with_unit(s: &str) -> Option<f64> {
276    let s = s.trim();
277    let mut end = 0;
278    let mut seen_dot = false;
279    let mut seen_exp = false;
280    for (i, c) in s.char_indices() {
281        let ok = match c {
282            '0'..='9' => true,
283            // 负号/正号只允许出现在开头或指数符号之后
284            '-' | '+' => i == 0 || matches!(s.as_bytes().get(i - 1), Some(b'e') | Some(b'E')),
285            '.' if !seen_dot && !seen_exp => {
286                seen_dot = true;
287                true
288            },
289            'e' | 'E' if !seen_exp && i > 0 => {
290                seen_exp = true;
291                true
292            },
293            _ => false,
294        };
295        if !ok {
296            break;
297        }
298        end = i + c.len_utf8();
299    }
300    if end == 0 {
301        return None;
302    }
303    s[..end].parse().ok()
304}

306/// 解析逗号分隔的通道号列表。
307///
308/// 偏离 Python 基线：Python 的 `str2ids` 要求通道号严格递增，非递增项被静默跳过。
309/// 该过滤会吃掉列表中的 `0`（未配置占位）以及紧随其后的方向标志位，
310/// 例如 `TA_CHNS=41,42,43,0,1` 会被压缩为 `[41,42,43]`，方向标志整体丢失。
311/// 此处按原文逐项解析，不做递增过滤。
312fn parse_num_list(s: &str) -> Vec<usize> {
313    s.split(',')
314        .filter_map(|p| {
315            let p = p.trim();
316            if p.is_empty() {
317                None
318            } else {
319                parse_num_with_unit(p).map(|v| v.max(0.0) as usize)
320            }
321        })
322        .collect()
323}

325/// 解析 `TV_CHNS=Ua,Ub,Uc,Un[,Ul]` 形式的电压通道组。
326fn parse_acv(s: &str) -> AcvChn {
327    let v = parse_num_list(s);
328    AcvChn {
329        ua_idx: v.first().copied().unwrap_or(0),
330        ub_idx: v.get(1).copied().unwrap_or(0),
331        uc_idx: v.get(2).copied().unwrap_or(0),
332        un_idx: v.get(3).copied().unwrap_or(0),
333        ul_idx: v.get(4).copied().unwrap_or(0),
334    }
335}

337/// 解析线路 `TA_CHNS=Ia,Ib,Ic,Io[,DIR]`（规范 §1.4，5 字段：含 `Io`）。
338///
339/// `DIR` 为可选尾项：`1` = 正方向（流向线路），`-1` = 反方向，缺省 `1`。
340fn parse_line_ta(s: &str, bran_idx: usize) -> AccBran {
341    let parts: Vec<&str> = s.split(',').map(|p| p.trim()).collect();
342    let num = |i: usize| -> usize {
343        parts
344            .get(i)
345            .and_then(|p| parse_num_with_unit(p))
346            .map(|v| v.max(0.0) as usize)
347            .unwrap_or(0)
348    };
349    AccBran {
350        bran_idx,
351        ia_idx: num(0),
352        ib_idx: num(1),
353        ic_idx: num(2),
354        in_idx: num(3),
355        dir: parse_dir_field(parts.get(4)),
356        dir_raw: parts.get(4).map(|s| s.to_string()).unwrap_or_default(),
357    }
358}

360/// 解析变压器 `TA_Id_#N=Ia,Ib,Ic,极性符号`（规范 §1.5，4 字段：**无** `Io`）。
361///
362/// 与线路 `TA_CHNS` 的字段数不同：变压器侧第 4 项已是极性符号，不是零序电流通道。
363/// 混用同一个解析器会把极性符号误读成 `in_idx`（Python 基线即如此）。
364fn parse_transformer_ta(s: &str, bran_idx: usize) -> AccBran {
365    let parts: Vec<&str> = s.split(',').map(|p| p.trim()).collect();
366    let num = |i: usize| -> usize {
367        parts
368            .get(i)
369            .and_then(|p| parse_num_with_unit(p))
370            .map(|v| v.max(0.0) as usize)
371            .unwrap_or(0)
372    };
373    AccBran {
374        bran_idx,
375        ia_idx: num(0),
376        ib_idx: num(1),
377        ic_idx: num(2),
378        in_idx: 0,
379        dir: parse_dir_field(parts.get(3)),
380        dir_raw: parts.get(3).map(|s| s.to_string()).unwrap_or_default(),
381    }
382}

384/// 解析方向/极性字段：`-1` → `-1`，其他（含缺省、空串、非法值）→ `1`。
385fn parse_dir_field(raw: Option<&&str>) -> i32 {
386    match raw.map(|s| s.trim()) {
387        Some(s) if !s.is_empty() && parse_num_with_unit(s).unwrap_or(1.0) < 0.0 => -1,
388        _ => 1,
389    }
390}

392/// 解析 `DEV_ID=idx,name`，返回 `(idx, name)`。
393///
394/// `name` 中可能含逗号，只按首个逗号切分。缺失 idx 时回落到 `fallback_idx`（节序号）。
395fn split_dev_id(raw: &str, fallback_idx: usize) -> (usize, String) {
396    match raw.split_once(',') {
397        Some((idx, name)) => {
398            let idx = idx
399                .trim()
400                .parse::<usize>()
401                .ok()
402                .filter(|v| *v > 0)
403                .unwrap_or(fallback_idx);
404            (idx, name.trim().to_string())
405        },
406        None => {
407            let t = raw.trim();
408            // 无逗号：整串可能是纯 idx，也可能是纯名字
409            match t.parse::<usize>() {
410                Ok(v) if v > 0 => (v, String::new()),
411                _ => (fallback_idx, t.to_string()),
412            }
413        },
414    }
415}

417/// 解析 `REACTOR`：`NO`（不区分大小写）表示无并联电抗器 → `None`。
418///
419/// 样本中还存在 `-1(Ω)` 的写法，同样按"无"处理：负补偿电抗无物理意义。
420fn parse_reactor(raw: &str) -> Option<f64> {
421    let t = raw.trim();
422    if t.is_empty() || t.eq_ignore_ascii_case("NO") {
423        return None;
424    }
425    match parse_num_with_unit(t) {
426        Some(v) if v < 0.0 => None,
427        other => other,
428    }
429}

431/// 解析 `RX=R1,X1,R0,X0`
432fn parse_impedance(s: &str) -> Impedance {
433    let v: Vec<f64> = s
434        .split(',')
435        .map(|p| parse_num_with_unit(p).unwrap_or(0.0))
436        .collect();
437    Impedance {
438        r1: v.first().copied().unwrap_or(0.0),
439        x1: v.get(1).copied().unwrap_or(0.0),
440        r0: v.get(2).copied().unwrap_or(0.0),
441        x0: v.get(3).copied().unwrap_or(0.0),
442    }
443}

445/// 解析 `CG=C1,G1,C0,G0`
446///
447/// 注意位置序：规范 §1.4 为 `C1,G1,C0,G0`（容/导交替），
448/// **不是** DMF `CG` 元素属性所暗示的 `c1,c0,g1,g0`。
449/// Python 基线按 `c1,c0,g1,g0` 读取，会把零序电容与正序电导对调。
450fn parse_capacitance(s: &str) -> Capacitance {
451    let v: Vec<f64> = s
452        .split(',')
453        .map(|p| parse_num_with_unit(p).unwrap_or(0.0))
454        .collect();
455    Capacitance {
456        c1: v.first().copied().unwrap_or(0.0),
457        g1: v.get(1).copied().unwrap_or(0.0),
458        c0: v.get(2).copied().unwrap_or(0.0),
459        g0: v.get(3).copied().unwrap_or(0.0),
460    }
461}

463/// 解析 `MRX=MR,MX`
464fn parse_mutual(s: &str) -> MutualInductance {
465    let v: Vec<f64> = s
466        .split(',')
467        .map(|p| parse_num_with_unit(p).unwrap_or(0.0))

469    MutualInductance {

476/// 解析 `{H,M,L}_PARAM=接线方式, 一次额定电压(kV), 分支数`
477///
478/// 第一字段是绕组接线组别**字符串代号**（`Y` / `Y12` / `D11` / `yn0`），不是数值。
479fn parse_winding_param(s: &str) -> (String, f64, usize) {
480    let parts: Vec<&str> = s.split(',').map(|p| p.trim()).collect();
481    let wg = parts.first().copied().unwrap_or("").to_string();
482    let v_rtg = parts
483        .get(1)
484        .and_then(|p| parse_num_with_unit(p))
485        .unwrap_or(0.0);
486    let bran_num = parts
487        .get(2)
488        .and_then(|p| parse_num_with_unit(p))
489        .map(|v| v.max(0.0) as usize)
490        .unwrap_or(0);
491    (wg, v_rtg, bran_num)
492}

494/// 按 `WINDING_NUM` 确定应构建的绕组侧。
495///
496/// 规范 §1.5：`3` = 三卷变（高/中/低），`2` = 两卷变（高/低，**缺中压侧**），
497/// `1` = 自耦变。
498///
499/// 自耦变（`1`）当前按两卷变（高/低）降级处理 —— 其专有的 `TA_Idcw_#n`
500/// 公共绕组电流通道尚未建模。
501/// TODO: 建模 `TA_Idcw_#1..#5` 后为自耦变返回独立的绕组布局。
502fn winding_layout(winding_num: usize) -> &'static [WindingLocation] {
503    match winding_num {
504        3 => &[
505            WindingLocation::High,
506            WindingLocation::Medium,
507            WindingLocation::Low,
508        ],
509        // 2（两卷变）与 1（自耦变）均为高/低两侧
510        _ => &[WindingLocation::High, WindingLocation::Low],
511    }
512}

514/// 构建单个绕组侧。该侧既无 `_PARAM` 也无任何 `TA_Id_#N` 时返回 `None`，
515/// 避免为不存在的中压侧凭空造出空绕组。
516fn build_winding(section: &Section, loc: WindingLocation) -> Option<TransformerWinding> {
517    let p = loc.inf_prefix();
518
519    let param = section.get(&format!("{}_PARAM", p));
520    let mut currents = Vec::new();
521    for (n, ta_idx) in loc.ta_id_indices().iter().enumerate() {
522        if let Some(raw) = section.get(&format!("TA_Id_#{}", ta_idx)) {
523            let bran = parse_transformer_ta(raw, n + 1);
524            if !bran.is_empty() {
525                currents.push(bran);
526            }
527        }
528    }
529
530    if param.is_none() && currents.is_empty() {
531        return None;
532    }
533
534    let (wg, v_rtg, bran_num) = param.map(parse_winding_param).unwrap_or_default();
535
536    Some(TransformerWinding {
537        location: loc.as_str().to_string(),
538        src_ref: String::new(),
539        v_rtg,
540        a_rtg: 0.0,
541        // `_PARAM` 缺失时用实际解析出的分支数兜底
542        bran_num: if bran_num > 0 {
543            bran_num
544        } else {
545            currents.len()
546        },
547        bus_id: String::new(),
548        wg,
549        acv: section
550            .get(&format!("{}_TV_CHNS", p))
551            .map(parse_acv)
552            .unwrap_or_default(),
553        currents,
554        igap: Igap::default(),
555        ta_zs: section
556            .get(&format!("{}_TA_ZS", p))
557            .and_then(parse_num_with_unit)
558            .map(|v| v.max(0.0) as usize)
559            .unwrap_or(0),
560        ta_zs_gap: section
561            .get(&format!("{}_TA_ZS_GAP", p))
562            .and_then(parse_num_with_unit)
563            .map(|v| v.max(0.0) as usize)
564            .unwrap_or(0),
565        sta_chns: section
566            .get(&format!("{}_STATUS_CHNS", p))
567            .map(parse_num_list)
568            .unwrap_or_default(),
569    })
570}

572/// 解析 N 元组数值（逗号分隔，逐项 `parse_num_with_unit`，不足补 0）。
573fn parse_tuple(s: Option<&str>, n: usize) -> Vec<f64> {
574    let mut out = vec![0.0; n];
575    if let Some(s) = s {
576        for (i, part) in s.split(',').take(n).enumerate() {
577            if let Some(v) = parse_num_with_unit(part) {
578                out[i] = v;
579            }
580        }
581    }
582    out
583}
584
585/// 解析 N 元组通道号（逗号分隔，逐项转 `usize`，不足补 0）。
586fn parse_tuple_usize(s: Option<&str>, n: usize) -> Vec<usize> {
587    let mut out = vec![0usize; n];
588    if let Some(s) = s {
589        for (i, part) in s.split(',').take(n).enumerate() {
590            if let Some(v) = parse_num_with_unit(part) {
591                out[i] = v.max(0.0) as usize;
592            }
593        }
594    }
595    out
596}

598/// 解析 §1.6 `BRANCH_NUM=z1, z2, z3`
599fn parse_branch_num(s: Option<&str>) -> BranchNum {
600    let v = parse_tuple_usize(s, 3);
601    BranchNum {
602        z1: v[0],
603        z2: v[1],
604        z3: v[2],
605    }
606}

608/// 解析 §1.6 `Ufe=额定励磁电压, 额定空载励磁电压`
609fn parse_ufe(s: Option<&str>) -> Ufe {
610    let v = parse_tuple(s, 2);
611    Ufe {
612        rated: v[0],
613        no_load: v[1],
614    }
615}

619    let v = parse_tuple(s, 4);
620    SyncReactance {
621        xd: v[0],
622        xq: v[1],
623        xd_prime: v[2],
624        xs: v[3],
625    }

635    }
