1//! 由 Config + 设备组生成 INF。
2
3use cbase::equipment::EquipmentGroup;
4use cfg::Config;
5
6use super::builder;
7use super::section::{InfFile, Section, SectionKind};
8
9/// 由 Config + 设备组生成 INF
10pub fn from_config(cfg: &Config, equipment: Option<&EquipmentGroup>) -> InfFile {
11    let mut sections = Vec::new();
12
13    // File_Description
14    sections.push(builder::build_file_description(cfg));
15
16    // 模拟通道节
17    for analog in &cfg.analogs {
18        sections.push(builder::build_analog_section(analog));
19    }
20
21    // 状态通道节
22    for status in &cfg.statuses {
23        sections.push(builder::build_status_section(status));
24    }
25
26    // 参数段
27    if !cfg.analogs.is_empty() {
28        let mut fields = Vec::new();
29        for analog in &cfg.analogs {
30            let param = builder::build_analog_parameter(analog);
31            fields.push((format!("CHNL_INFO_#{}", analog.index), param));
32        }
33        sections.push(Section {
34            area: "ZYHD".into(),
35            kind: SectionKind::AnalogChannelsParameter,
36            index: 0,
37            fields,
38            raw: String::new(),
39        });
40    }
41
42    if !cfg.statuses.is_empty() {
43        let mut fields = Vec::new();
44        for status in &cfg.statuses {
45            let param = builder::build_status_parameter(status);
46            fields.push((format!("CHNL_INFO_#{}", status.index), param));
47        }
48        sections.push(Section {
49            area: "ZYHD".into(),
50            kind: SectionKind::StatusChannelsParameter,
51            index: 0,
52            fields,
53            raw: String::new(),
54        });
55    }
56
57    // 设备节
58    if let Some(eg) = equipment {
59        for bus in &eg.buses {
60            sections.push(builder::build_bus_section(bus));
61        }
62        for line in &eg.lines {
63            sections.push(builder::build_line_section(line));
64        }
65        for transformer in &eg.transformers {
66            sections.push(builder::build_transformer_section(transformer));
67        }
68        for generator in &eg.generators {
69            sections.push(builder::build_generator_section(generator));
70        }
71        for exciter in &eg.exciters {
72            sections.push(builder::build_exciter_section(exciter));
73        }
74    }
75
76    InfFile { sections }
77}
