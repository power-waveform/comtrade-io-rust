1//! 采样信息模型
2
3/// 额定频率缺省值（Hz）
4///
5/// 当 CFG 未给出或给出非法（`<= 0` / 非有限）额定频率时使用，
6/// 对齐 Python 基线 `freq if freq else 50`。
7pub const DEFAULT_NOMINAL_FREQ: f64 = 50.0;
8
9/// 采样段
10///
11/// `samp_rate` / `end_point` 来自 CFG 文件本身；`start_point` / `count` /
12/// `cycle_point_num` 是重算时才产生的派生字段，CFG 不携带，故解析出的段一律为 `None`
13/// （对齐 Python 基线中这三个字段的 `| None` 类型与 `default=None`）。
14#[derive(Debug, Clone, Copy)]
15pub struct Segment {
16    /// 该段采样率（Hz）
17    pub samp_rate: f64,
18    /// 该段累计结束点
19    ///
20    /// 即"该段最后一个采样点的 1-based 序号"，数值上等于半开区间
21    /// `[start_point, end_point)` 的右端点（0-based）——两种表述取值相同，
22    /// 对齐 CFG 的 `endsamp` 字段语义。
23    pub end_point: usize,
24    /// 该段起始采样点号（0-based，半开区间 `[start_point, end_point)` 的左端点）
25    ///
26    /// 派生字段，仅 `recalculate_segments`（dat）填充。
27    pub start_point: Option<usize>,
28    /// 该段采样点数（`end_point - start_point`）
29    ///
30    /// 派生字段，仅 `recalculate_segments`（dat）填充。
31    pub count: Option<usize>,
32    /// 该段每周波采样点数（`samp_rate / 额定频率`）
33    ///
34    /// 派生字段，仅 `recalculate_segments`（dat）按额定频率填充。
35    pub cycle_point_num: Option<f64>,
36}
37
38impl Segment {
39    /// 构造 CFG 原生段：仅采样率与结束点，派生字段留空
40    pub fn new(samp_rate: f64, end_point: usize) -> Self {
41        Segment {
42            samp_rate,
43            end_point,
44            start_point: None,
45            count: None,
46            cycle_point_num: None,
47        }
48    }
49
50    /// 构造重算段：附带 `start_point` / `count` / `cycle_point_num` 三个派生字段
51    ///
52    /// `start_point` 与 `end_point` 构成半开区间 `[start_point, end_point)`，
53    /// `count` 由二者之差算出，`cycle_point_num` 为 `samp_rate / nominal_freq`。
54    pub fn derived(
55        samp_rate: f64,
56        start_point: usize,
57        end_point: usize,
58        nominal_freq: f64,
59    ) -> Self {
60        Segment {
61            samp_rate,
62            end_point,
63            start_point: Some(start_point),
64            count: Some(end_point.saturating_sub(start_point)),
65            cycle_point_num: Some(samp_rate / nominal_freq),
66        }
67    }
68}
69
70/// 采样信息
71#[derive(Debug, Clone, Default)]
72pub struct Sampling {
73    /// 标称频率（Hz）
74    pub freq: f64,
75    /// 采样段列表
76    pub segments: Vec<Segment>,
77}
78
79impl Sampling {
80    /// 序列化为 CFG 文本（频率行 + 段数行 + 各段行）
81    pub fn to_cfg_text(&self) -> String {
82        let mut lines = Vec::new();
83        lines.push(format!("{}", self.freq));
84        lines.push(format!("{}", self.segments.len()));
85        for seg in &self.segments {
86            lines.push(format!("{:.0},{:.0}", seg.samp_rate, seg.end_point));
87        }
88        lines.join("\n")
89    }
