3use super::version::Version;
4use cbase::error::{Error, FileRole, Result};

6/// 文件头
7#[derive(Debug, Clone)]
8pub struct Header {
9    /// 站名（录波装置所在变电站/站点名称）
10    pub station: String,
11    /// 录波装置名称
12    pub recorder: String,
13    /// 录波数据标准版本（1991/1999/2001/2008/2013/2017）
14    pub version: Version,
15}
16
17impl Header {
18    /// 从 CFG 首行解析文件头（3 字段，缺失/非法版本回落 1991）
19    pub fn from_line(line: &str) -> Header {
20        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
21        let station = parts.first().unwrap_or(&"").to_string();
22        let recorder = parts.get(1).unwrap_or(&"").to_string();
23        // 六个版本号全部识别；头部无版本字段或版本号无法识别时回落 1991
24        // （对齐 Python `Version.from_value(str_arr[2], Version.V1991)`）。
25        // 原实现只匹配 "1999"，导致 2001/2008/2013/2017 的文件被静默判成 1991。
26        let version = parts
27            .get(2)
28            .and_then(|s| Version::parse(s))
29            .unwrap_or_default();
30        Header {
31            station,
32            recorder,
33            version,
34        }
35    }
36
37    /// 序列化为 CFG 首行
38    pub fn to_line(&self) -> String {
39        format!(
40            "{},{},{}",
41            self.station,
42            self.recorder,
43            self.version.as_str()
44        )
45    }
46}

48/// 通道计数
49#[derive(Debug, Clone, Copy)]
50pub struct ChannelCount {
51    /// 通道总数（模拟量 + 状态量）
52    pub total: usize,
53    /// 模拟量通道数
54    pub analog: usize,
55    /// 状态量通道数
56    pub status: usize,
57}
58
59impl ChannelCount {
60    pub(crate) fn from_line(line: &str) -> Result<ChannelCount> {
61        let parts: Vec<&str> = line.split(',').map(|s| s.trim()).collect();
62        if parts.len() < 3 {
63            return Err(Error::parse(FileRole::Cfg, 2, "通道数行格式错误"));
64        }
65        let total = parts[0]
66            .parse::<usize>()
67            .map_err(|_| Error::number(FileRole::Cfg, parts[0].to_string()))?;
68        let analog = extract_digits(parts.get(1).unwrap_or(&"0"))
69            .map_err(|_| Error::number(FileRole::Cfg, parts[1].to_string()))?;
70        let status = extract_digits(parts.get(2).unwrap_or(&"0"))
71            .map_err(|_| Error::number(FileRole::Cfg, parts[2].to_string()))?;
72        Ok(ChannelCount {
73            total,
74            analog,
75            status,
76        })
77    }
78
79    pub(crate) fn to_line(self) -> String {
80        format!("{},{}A,{}D", self.total, self.analog, self.status)
81    }
82}

84/// 从 "96A" / "192D" 类字符串中提取数字
85fn extract_digits(s: &str) -> std::result::Result<usize, std::num::ParseIntError> {
86    let digits: String = s.chars().filter(|c| c.is_ascii_digit()).collect();
87    if digits.is_empty() {
88        return Ok(0);
89    }
90    digits.parse()
