1//! 变压器与绕组元素属性解析
2
3use cbase::equipment::{AcvChn, Igap, Transformer, TransformerWinding};
4
5use super::common::{attr_f64, attr_str, attr_usize};
6
7pub fn parse_transformer_attrs(attrs: &[(&str, &str)]) -> Transformer {
8    Transformer {
9        idx: attr_usize(attrs, "idx"),
10        name: attr_str(attrs, "trm_name"),
11        src_ref: attr_str(attrs, "srcRef"),
12        sys_id: String::new(),
13        object_type: String::new(),
14        pwr_rtg: attr_f64(attrs, "pwrRtg", 0.0),
15        winding_num: 0,
16        ta_self_comp: String::new(),
17        transformer_uuid: attr_str(attrs, "transformer_uuid"),
18        windings: Vec::new(),
19        oth_achns: Vec::new(),
20        ana_chns: Vec::new(),
21        sta_chns: Vec::new(),
22    }
23}
24
25pub fn parse_winding_attrs(attrs: &[(&str, &str)]) -> TransformerWinding {
26    TransformerWinding {
27        location: attr_str(attrs, "location"),
28        src_ref: attr_str(attrs, "srcRef"),
29        v_rtg: attr_f64(attrs, "VRtg", 0.0),
30        a_rtg: attr_f64(attrs, "ARtg", 0.0),
31        bran_num: attr_usize(attrs, "bran_num"),
32        bus_id: attr_str(attrs, "bus_ID"),
33        // 偏离 Python 基线：wG 是绕组接线组别代号（如 y0 / yn0），
34        // 原 Rust 实现声明为 f64 并 parse().unwrap_or(0.0)，使全部组别静默归零。
35        wg: attr_str(attrs, "wG"),
36        acv: AcvChn::default(),
37        currents: Vec::new(),
38        igap: Igap::default(),
39        ta_zs: 0,
40        ta_zs_gap: 0,
41        sta_chns: Vec::new(),
42    }
43}
