1//! 发电机 / 励磁机元素属性解析（§1.6 / §1.7）
2
3use cbase::equipment::{
4    AccBran, AcvChn, BranchNum, Exciter, Generator, SyncReactance, Ufe, UfeChns, UnChns,
5};
6
7use super::common::{attr, attr_f64, attr_str, attr_usize};
8
9/// 解析发电机属性（§1.6）。DMF 对发电机/励磁机的元素与属性命名规范未定义，
10/// 此处按与现有 Bus/Line 风格一致的属性名读取。
11pub fn parse_generator_attrs(attrs: &[(&str, &str)]) -> Generator {
12    Generator {
13        idx: attr_usize(attrs, "idx"),
14        name: attr_str(attrs, "gen_name"),
15        src_ref: attr_str(attrs, "srcRef"),
16        sys_id: attr_str(attrs, "sys_ID"),
17        trm_id: attr_str(attrs, "trm_ID"),
18        object_type: attr_str(attrs, "object_type"),
19        freq: attr_f64(attrs, "freq", 0.0),
20        capacity: attr_f64(attrs, "capacity", 0.0),
21        factor: attr_f64(attrs, "factor", 0.0),
22        v1: attr_f64(attrs, "V1", 0.0),
23        branch_num: BranchNum {
24            z1: attr_usize(attrs, "branch_z1"),
25            z2: attr_usize(attrs, "branch_z2"),
26            z3: attr_usize(attrs, "branch_z3"),
27        },
28        rotor_i: attr_f64(attrs, "rotor_I", 0.0),
29        rotor_v2: attr_f64(attrs, "rotor_V2", 0.0),
30        ufe: Ufe {
31            rated: attr_f64(attrs, "ufe_rated", 0.0),
32            no_load: attr_f64(attrs, "ufe_no_load", 0.0),
33        },
34        x: SyncReactance {
35            xd: attr_f64(attrs, "xd", 0.0),
36            xq: attr_f64(attrs, "xq", 0.0),
37            xd_prime: attr_f64(attrs, "xd_prime", 0.0),
38            xs: attr_f64(attrs, "xs", 0.0),
39        },
40        excitation_mode: attr(attrs, "excitation_mode")
41            .and_then(|v| v.trim().parse::<i32>().ok())
42            .unwrap_or(0),
43        igt_dir: attr(attrs, "igt_dir")
44            .and_then(|v| v.trim().parse::<i32>().ok())
45            .unwrap_or(0),
46        acv: AcvChn::default(),
47        ta: AccBran::default(),
48        ta_z1: AccBran::default(),
49        ta_z2: AccBran::default(),
50        ta_z3: AccBran::default(),
51        ufe_chns: UfeChns {
52            ufe: attr_usize(attrs, "ufe_chn"),
53            pos: attr_usize(attrs, "pos_ufe_chn"),
54            neg: attr_usize(attrs, "neg_ufe_chn"),
55        },
56        ife_chn: attr_usize(attrs, "ife_chn"),
57        un_chns: UnChns {
58            terminal: attr_usize(attrs, "un_terminal"),
59            neutral: attr_usize(attrs, "un_neutral"),
60            longitudinal: attr_usize(attrs, "un_longitudinal"),
61        },
62        ta_ido_chn: attr_usize(attrs, "ta_ido_chn"),
63        oth_achns: Vec::new(),
64        sta_chns: Vec::new(),
65    }
66}

68/// 解析励磁机属性（§1.7）。
69pub fn parse_exciter_attrs(attrs: &[(&str, &str)]) -> Exciter {
70    Exciter {
71        idx: attr_usize(attrs, "idx"),
72        name: attr_str(attrs, "exc_name"),
73        src_ref: attr_str(attrs, "srcRef"),
74        sys_id: attr_str(attrs, "sys_ID"),
75        pwr_id: attr_str(attrs, "pwr_ID"),
76        object_type: attr_str(attrs, "object_type"),
77        freq: attr_f64(attrs, "freq", 0.0),
78        v1: attr_f64(attrs, "V1", 0.0),
79        acv: AcvChn::default(),
80        ta: AccBran::default(),
81        ta_z: AccBran::default(),
82        oth_achns: Vec::new(),
83        sta_chns: Vec::new(),
84    }
