6/// 属性查表（大小写敏感，与 XML 规范一致）
7pub fn attr<'a>(attrs: &'a [(&str, &str)], name: &str) -> Option<&'a str> {
8    attrs.iter().find(|(k, _)| *k == name).map(|(_, v)| *v)
9}
10
11/// 读取整型属性，缺失/非法/空串均为 0
12pub fn attr_usize(attrs: &[(&str, &str)], name: &str) -> usize {
13    attr(attrs, name)
14        .and_then(|v| v.trim().parse().ok())
15        .unwrap_or(0)
16}
17
18/// 读取浮点属性，缺失/非法/空串均为 `default`
19pub fn attr_f64(attrs: &[(&str, &str)], name: &str, default: f64) -> f64 {
20    attr(attrs, name)
21        .and_then(|v| v.trim().parse().ok())
22        .unwrap_or(default)
23}

25/// 读取字符串属性
26pub fn attr_str(attrs: &[(&str, &str)], name: &str) -> String {
27    attr(attrs, name).unwrap_or("").to_string()
28}

40pub fn parse_acc_attrs(attrs: &[(&str, &str)]) -> AccBran {
41    // 偏离 Python 基线：读 bran_idx 写 idx，导致二次解析时 idx 归零。此处读写统一为 bran_idx。
42    let dir_raw = attr_str(attrs, "dir");
43    AccBran {
44        bran_idx: attr_usize(attrs, "bran_idx"),
45        ia_idx: attr_usize(attrs, "ia_idx"),
46        ib_idx: attr_usize(attrs, "ib_idx"),
47        ic_idx: attr_usize(attrs, "ic_idx"),
48        in_idx: attr_usize(attrs, "in_idx"),
49        dir: parse_dir(&dir_raw),
50        dir_raw,
51    }
52}

54/// 归一化方向标志。
55///
56/// DMF 用 `POS` / `NEG`（样本中大小写不一），INF 用 `1` / `-1`（规范 §1.4）。
57/// 缺省为正方向 `1`。
58pub fn parse_dir(s: &str) -> i32 {
59    let t = s.trim();
60    if t.is_empty() {
61        return 1;
62    }
63    if let Ok(v) = t.parse::<i32>() {
64        return if v < 0 { -1 } else { 1 };
65    }
66    match t.to_ascii_lowercase().as_str() {
67        "neg" | "negative" | "-" => -1,
68        _ => 1,
69    }
70}

72pub fn parse_analog_attrs(attrs: &[(&str, &str)]) -> DmfAnalogChannel {
73    let mut ch = DmfAnalogChannel::default();
74    let mut has_au = false;
75    for (name, value) in attrs {
76        match *name {
77            "idx_cfg" => ch.idx_cfg = value.parse().unwrap_or(0),
78            "idx_org" => ch.idx_org = value.parse().unwrap_or(0),
79            "type" => ch.ch_type = value.to_string(),
80            "flag" => ch.flag = value.to_string(),
81            "freq" => ch.freq = value.parse().unwrap_or(50.0),
82            "au" => {
83                has_au = true;
84                ch.au = value.parse().unwrap_or(0.0)
85            },
86            "bu" => ch.bu = value.parse().unwrap_or(0.0),
87            "sIUnit" => ch.unit = value.to_string(),
88            "multiplier" => ch.multiplier = value.parse().unwrap_or(1.0),
89            "primary" => ch.primary = value.parse().unwrap_or(1.0),
90            "secondary" => ch.secondary = value.parse().unwrap_or(1.0),
91            "ps" => ch.ps = value.to_string(),
92            // 兼容早期 Python 写出的 idx_rl 与当前样本使用的 idx_rlt。
93            "idx_rlt" | "idx_rl" => ch.idx_rlt = value.parse().unwrap_or(0),
94            "ph" => ch.ph = value.to_string(),
95            _ => {},
96        }
97    }
98    // DMF 早期版本可能省略 au。交流模拟量的标准默认值为 1；
99    // 显式写入（包括显式 0）的值必须保留。
100    if !has_au && is_ac_channel_type(&ch.ch_type) {
101        ch.au = 1.0;
102    }
103    ch
104}

106pub fn is_ac_channel_type(kind: &str) -> bool {
107    matches!(
108        kind.trim().to_ascii_lowercase().as_str(),
109        "a" | "ac" | "acv" | "acc" | "acvoltage" | "ac current" | "acccurrent" | "accurrent"
110    )
111}
